# 2.1 Create WebSocket Client Layer

## Objective
Build a robust WebSocket client in the wos-hmi module that UI controllers can use to communicate with the backend API.

## Duration
**6-8 hours**

## Prerequisites
- Completed Phase 1 (Backend API server is running and tested)
- Understanding of Java-WebSocket library
- Familiarity with CompletableFuture and async patterns

---

## Architecture

The client layer consists of:

1. **BotApiClient** - Main client interface
2. **Connection Manager** - Handles connection/reconnection
3. **Message Handler** - Parses responses and events
4. **Request Tracker** - Correlates requests with responses

```
┌─────────────────────┐
│   UI Controllers    │
└──────────┬──────────┘
           │
           ▼
┌─────────────────────┐
│   BotApiClient      │
│  - sendCommand()    │
│  - onEvent()        │
└──────────┬──────────┘
           │
           ▼
┌─────────────────────┐
│  WebSocketClient    │
│  (Java-WebSocket)   │
└──────────┬──────────┘
           │ ws://
           ▼
┌─────────────────────┐
│  Backend Server     │
└─────────────────────┘
```

## Implementation

### 1. Add Dependency

**File:** `wos-hmi/pom.xml`

Add Java-WebSocket library:

```xml
<dependency>
    <groupId>org.java-websocket</groupId>
    <artifactId>Java-WebSocket</artifactId>
    <version>1.5.4</version>
</dependency>

<!-- Keep wos-ot for DTOs -->
<dependency>
    <groupId>cl.camodev</groupId>
    <artifactId>wos-ot</artifactId>
    <version>${revision}</version>
</dependency>

<!-- Add wos-api for protocol classes -->
<dependency>
    <groupId>cl.camodev</groupId>
    <artifactId>wos-api</artifactId>
    <version>${revision}</version>
    <classifier>client</classifier> <!-- Only client code, not server -->
</dependency>
```

### 2. Create BotApiClient

**File:** `wos-hmi/src/main/java/cl/camodev/wosbot/client/BotApiClient.java`

```java
package cl.camodev.wosbot.client;

import cl.camodev.wosbot.api.protocol.Command;
import cl.camodev.wosbot.api.protocol.Event;
import cl.camodev.wosbot.api.protocol.Response;
import cl.camodev.wosbot.ot.*;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.java_websocket.client.WebSocketClient;
import org.java_websocket.handshake.ServerHandshake;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.net.URI;
import java.util.*;
import java.util.concurrent.*;
import java.util.function.Consumer;

/**
 * WebSocket client for communicating with WosBot backend API.
 */
public class BotApiClient {
    private static final Logger logger = LoggerFactory.getLogger(BotApiClient.class);
    private static final int COMMAND_TIMEOUT_MS = 30000;

    private final String serverUrl;
    private WebSocketClient webSocketClient;
    private final ObjectMapper objectMapper = new ObjectMapper();

    // Track pending requests (id → future)
    private final Map<String, CompletableFuture<Response>> pendingRequests = new ConcurrentHashMap<>();

    // Event handlers (eventName → list of handlers)
    private final Map<String, List<Consumer<Event>>> eventHandlers = new ConcurrentHashMap<>();

    // Connection state
    private final CompletableFuture<Void> connectionFuture = new CompletableFuture<>();
    private boolean connected = false;

    public BotApiClient(String serverUrl) {
        this.serverUrl = serverUrl;
    }

    // ==================== Connection Management ====================

    /**
     * Connect to backend server.
     */
    public CompletableFuture<Void> connect() {
        logger.info("Connecting to backend at {}", serverUrl);

        try {
            webSocketClient = new WebSocketClient(new URI(serverUrl)) {
                @Override
                public void onOpen(ServerHandshake handshake) {
                    handleOpen(handshake);
                }

                @Override
                public void onMessage(String message) {
                    handleMessage(message);
                }

                @Override
                public void onClose(int code, String reason, boolean remote) {
                    handleClose(code, reason, remote);
                }

                @Override
                public void onError(Exception ex) {
                    handleError(ex);
                }
            };

            webSocketClient.connect();
            return connectionFuture;

        } catch (Exception e) {
            logger.error("Failed to create WebSocket client", e);
            connectionFuture.completeExceptionally(e);
            return connectionFuture;
        }
    }

    /**
     * Disconnect from server.
     */
    public void disconnect() {
        logger.info("Disconnecting from backend");
        if (webSocketClient != null) {
            webSocketClient.close();
        }
        connected = false;
    }

    /**
     * Check if connected to backend.
     */
    public boolean isConnected() {
        return connected && webSocketClient != null && webSocketClient.isOpen();
    }

    // ==================== Command Methods ====================

    /**
     * Start bot (all enabled profiles).
     */
    public CompletableFuture<Void> startBot() {
        return sendCommand("startBot", null).thenApply(r -> null);
    }

    /**
     * Stop bot (all profiles).
     */
    public CompletableFuture<Void> stopBot() {
        return sendCommand("stopBot", null).thenApply(r -> null);
    }

    /**
     * Pause all queues.
     */
    public CompletableFuture<Void> pauseAllQueues() {
        return sendCommand("pauseAllQueues", null).thenApply(r -> null);
    }

    /**
     * Resume all queues.
     */
    public CompletableFuture<Void> resumeAllQueues() {
        return sendCommand("resumeAllQueues", null).thenApply(r -> null);
    }

    /**
     * Pause specific queue.
     */
    public CompletableFuture<Void> pauseQueue(Long profileId) {
        return sendCommand("pauseQueue", Map.of("profileId", profileId)).thenApply(r -> null);
    }

    /**
     * Resume specific queue.
     */
    public CompletableFuture<Void> resumeQueue(Long profileId) {
        return sendCommand("resumeQueue", Map.of("profileId", profileId)).thenApply(r -> null);
    }

    /**
     * Get all profiles.
     */
    public CompletableFuture<List<DTOProfiles>> getProfiles() {
        return sendCommand("getProfiles", null)
            .thenApply(response -> {
                // Convert result to List<DTOProfiles>
                return objectMapper.convertValue(response.getResult(),
                    objectMapper.getTypeFactory().constructCollectionType(List.class, DTOProfiles.class));
            });
    }

    /**
     * Get global configuration.
     */
    public CompletableFuture<Map<String, String>> getGlobalConfig() {
        return sendCommand("getGlobalConfig", null)
            .thenApply(response -> {
                return objectMapper.convertValue(response.getResult(),
                    objectMapper.getTypeFactory().constructMapType(Map.class, String.class, String.class));
            });
    }

    /**
     * Get current stamina for profile.
     */
    public CompletableFuture<Integer> getCurrentStamina(Long profileId) {
        return sendCommand("getCurrentStamina", Map.of("profileId", profileId))
            .thenApply(response -> (Integer) response.getResult());
    }

    // ==================== Event Handlers ====================

    /**
     * Register handler for bot state changed events.
     */
    public void onBotStateChanged(Consumer<DTOBotState> handler) {
        registerEventHandler("botStateChanged", event -> {
            DTOBotState state = objectMapper.convertValue(event.getData(), DTOBotState.class);
            handler.accept(state);
        });
    }

    /**
     * Register handler for queue state changed events.
     */
    public void onQueueStateChanged(Consumer<DTOQueueState> handler) {
        registerEventHandler("queueStateChanged", event -> {
            DTOQueueState state = objectMapper.convertValue(event.getData(), DTOQueueState.class);
            handler.accept(state);
        });
    }

    /**
     * Register handler for stamina changed events.
     */
    public void onStaminaChanged(Consumer<Map<String, Object>> handler) {
        registerEventHandler("staminaChanged", event -> {
            Map<String, Object> data = objectMapper.convertValue(event.getData(), Map.class);
            handler.accept(data);
        });
    }

    /**
     * Register handler for log message events.
     */
    public void onLogMessage(Consumer<DTOLogMessage> handler) {
        registerEventHandler("logMessage", event -> {
            DTOLogMessage logMessage = objectMapper.convertValue(event.getData(), DTOLogMessage.class);
            handler.accept(logMessage);
        });
    }

    // ==================== Private Methods ====================

    /**
     * Send command to server and return future with response.
     */
    private CompletableFuture<Response> sendCommand(String commandName, Object params) {
        String id = UUID.randomUUID().toString();
        Command command = new Command(id, commandName, params);

        CompletableFuture<Response> future = new CompletableFuture<>();
        pendingRequests.put(id, future);

        // Set timeout
        CompletableFuture.delayedExecutor(COMMAND_TIMEOUT_MS, TimeUnit.MILLISECONDS)
            .execute(() -> {
                if (!future.isDone()) {
                    pendingRequests.remove(id);
                    future.completeExceptionally(new TimeoutException("Command timed out: " + commandName));
                }
            });

        try {
            String json = objectMapper.writeValueAsString(command);
            logger.debug("Sending command: {}", commandName);
            webSocketClient.send(json);
        } catch (Exception e) {
            pendingRequests.remove(id);
            future.completeExceptionally(e);
        }

        return future;
    }

    /**
     * Register event handler.
     */
    private void registerEventHandler(String eventName, Consumer<Event> handler) {
        eventHandlers.computeIfAbsent(eventName, k -> new ArrayList<>()).add(handler);
        logger.debug("Registered handler for event: {}", eventName);
    }

    /**
     * Handle WebSocket open.
     */
    private void handleOpen(ServerHandshake handshake) {
        logger.info("✅ Connected to backend");
        connected = true;
        connectionFuture.complete(null);
    }

    /**
     * Handle incoming message.
     */
    private void handleMessage(String message) {
        try {
            JsonNode node = objectMapper.readTree(message);

            if (node.has("id")) {
                // Response to command
                Response response = objectMapper.treeToValue(node, Response.class);
                handleResponse(response);
            } else if (node.has("event")) {
                // Event from server
                Event event = objectMapper.treeToValue(node, Event.class);
                handleEvent(event);
            } else {
                logger.warn("Unknown message format: {}", message);
            }

        } catch (Exception e) {
            logger.error("Failed to parse message: {}", message, e);
        }
    }

    /**
     * Handle command response.
     */
    private void handleResponse(Response response) {
        CompletableFuture<Response> future = pendingRequests.remove(response.getId());
        if (future != null) {
            if (response.isSuccess()) {
                future.complete(response);
            } else {
                future.completeExceptionally(new RuntimeException(response.getError()));
            }
        } else {
            logger.warn("Received response for unknown request: {}", response.getId());
        }
    }

    /**
     * Handle event from server.
     */
    private void handleEvent(Event event) {
        logger.debug("Received event: {}", event.getEvent());
        List<Consumer<Event>> handlers = eventHandlers.get(event.getEvent());
        if (handlers != null) {
            handlers.forEach(handler -> {
                try {
                    handler.accept(event);
                } catch (Exception e) {
                    logger.error("Error in event handler for {}", event.getEvent(), e);
                }
            });
        }
    }

    /**
     * Handle WebSocket close.
     */
    private void handleClose(int code, String reason, boolean remote) {
        logger.warn("Disconnected from backend. Code: {}, Reason: {}", code, reason);
        connected = false;

        // Fail all pending requests
        pendingRequests.values().forEach(future ->
            future.completeExceptionally(new RuntimeException("Connection closed"))
        );
        pendingRequests.clear();
    }

    /**
     * Handle WebSocket error.
     */
    private void handleError(Exception ex) {
        logger.error("WebSocket error", ex);
        if (!connectionFuture.isDone()) {
            connectionFuture.completeExceptionally(ex);
        }
    }
}
```

## Testing the Client

### 1. Simple Connection Test

**File:** `wos-hmi/src/test/java/cl/camodev/wosbot/client/BotApiClientTest.java`

```java
package cl.camodev.wosbot.client;

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

import java.util.concurrent.TimeUnit;

public class BotApiClientTest {

    @Test
    public void testConnection() throws Exception {
        // Assumes backend is running on localhost:8765
        BotApiClient client = new BotApiClient("ws://localhost:8765/bot");

        // Connect with 5 second timeout
        client.connect().get(5, TimeUnit.SECONDS);

        assertTrue(client.isConnected());

        client.disconnect();
    }

    @Test
    public void testGetProfiles() throws Exception {
        BotApiClient client = new BotApiClient("ws://localhost:8765/bot");
        client.connect().get(5, TimeUnit.SECONDS);

        var profiles = client.getProfiles().get(5, TimeUnit.SECONDS);

        assertNotNull(profiles);
        // Add more assertions based on your test data

        client.disconnect();
    }
}
```
