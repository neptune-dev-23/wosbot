# 1.3 Build Service Facade

## Objective
Create `BotApiService` facade class that wraps all 40+ backend service calls into a clean API that the WebSocket server can invoke.

## Duration
**6-8 hours**

## Prerequisites
- Completed [1.2 Implement WebSocket Server](1.2-websocket-server.md)
- Understanding of existing backend services (ServScheduler, ServConfig, etc.)
- List of all backend methods that need wrapping

---

## Service Method Inventory

Based on code analysis, these backend services need wrapping:

### ServScheduler (wos-serv)
- `startBot()`
- `stopBot()`
- `pauseBot()`
- `resumeBot()`
- `pauseQueue(Long profileId)`
- `resumeQueue(Long profileId)`
- `saveEmulatorPath(String key, String value)`

### ServConfig (wos-serv)
- `getGlobalConfig()` → `HashMap<String, String>`
- `saveGlobalConfig(String key, String value)`

### ServProfiles (wos-serv)
- `getProfiles()` → `List<DTOProfiles>`
- `saveProfile(Profile profile)`
- `deleteProfile(Long profileId)`
- `getProfileById(Long profileId)` → `Profile`

### ServTaskManager (wos-serv)
- `getTaskStatus(Long profileId)` → `DTOTaskState`
- `scheduleTask(...)`

### StaminaService (wos-serv)
- `getCurrentStamina(Long profileId)` → `int`
- `setStamina(Long profileId, int stamina)`

## Step-by-Step Implementation

### 1. Create BotApiService Class

**File:** `wos-api/src/main/java/cl/camodev/wosbot/api/service/BotApiService.java`

```java
package cl.camodev.wosbot.api.service;

import cl.camodev.wosbot.almac.entity.Profile;
import cl.camodev.wosbot.ot.DTOProfiles;
import cl.camodev.wosbot.ot.DTOTaskState;
import cl.camodev.wosbot.serv.impl.*;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.util.HashMap;
import java.util.List;

/**
 * Facade service that wraps all backend operations.
 * Provides a clean API for WebSocket server to call.
 */
public class BotApiService {
    private static final Logger logger = LoggerFactory.getLogger(BotApiService.class);

    // Singleton instances of backend services
    private final ServScheduler scheduler = ServScheduler.getServices();
    private final ServConfig config = ServConfig.getServices();
    private final ServProfiles profiles = ServProfiles.getServices();
    private final ServTaskManager taskManager = ServTaskManager.getServices();
    private final StaminaService staminaService = StaminaService.getServices();

    // ==================== Bot Control ====================

    /**
     * Start the bot (all enabled profiles).
     */
    public void startBot() {
        logger.info("API: Starting bot");
        scheduler.startBot();
    }

    /**
     * Stop the bot (all profiles).
     */
    public void stopBot() {
        logger.info("API: Stopping bot");
        scheduler.stopBot();
    }

    /**
     * Pause all queues.
     */
    public void pauseAllQueues() {
        logger.info("API: Pausing all queues");
        scheduler.pauseBot();
    }

    /**
     * Resume all queues.
     */
    public void resumeAllQueues() {
        logger.info("API: Resuming all queues");
        scheduler.resumeBot();
    }

    /**
     * Pause specific profile queue.
     */
    public void pauseQueue(Long profileId) {
        logger.info("API: Pausing queue for profile {}", profileId);
        scheduler.pauseQueue(profileId);
    }

    /**
     * Resume specific profile queue.
     */
    public void resumeQueue(Long profileId) {
        logger.info("API: Resuming queue for profile {}", profileId);
        scheduler.resumeQueue(profileId);
    }

    // ==================== Configuration ====================

    /**
     * Get all global configuration.
     */
    public HashMap<String, String> getGlobalConfig() {
        logger.debug("API: Getting global config");
        return config.getGlobalConfig();
    }

    /**
     * Save global configuration entry.
     */
    public void saveGlobalConfig(String key, String value) {
        logger.info("API: Saving config: {} = {}", key, value);
        config.saveGlobalConfig(key, value);
    }

    /**
     * Save emulator path.
     */
    public void saveEmulatorPath(String key, String value) {
        logger.info("API: Saving emulator path: {} = {}", key, value);
        scheduler.saveEmulatorPath(key, value);
    }

    // ==================== Profiles ====================

    /**
     * Get all profiles.
     */
    public List<DTOProfiles> getProfiles() {
        logger.debug("API: Getting all profiles");
        return profiles.getProfiles();
    }

    /**
     * Get profile by ID.
     */
    public Profile getProfileById(Long profileId) {
        logger.debug("API: Getting profile {}", profileId);
        return profiles.getProfileById(profileId);
    }

    /**
     * Save profile (create or update).
     */
    public void saveProfile(Profile profile) {
        logger.info("API: Saving profile: {}", profile.getName());
        profiles.saveProfile(profile);
    }

    /**
     * Delete profile.
     */
    public void deleteProfile(Long profileId) {
        logger.info("API: Deleting profile {}", profileId);
        profiles.deleteProfile(profileId);
    }

    // ==================== Task Management ====================

    /**
     * Get task status for profile.
     */
    public DTOTaskState getTaskStatus(Long profileId) {
        logger.debug("API: Getting task status for profile {}", profileId);
        return taskManager.getTaskStatus(profileId);
    }

    // ==================== Stamina ====================

    /**
     * Get current stamina for profile.
     */
    public int getCurrentStamina(Long profileId) {
        logger.debug("API: Getting stamina for profile {}", profileId);
        return staminaService.getCurrentStamina(profileId);
    }

    /**
     * Set stamina for profile.
     */
    public void setStamina(Long profileId, int stamina) {
        logger.info("API: Setting stamina for profile {} to {}", profileId, stamina);
        staminaService.setStamina(profileId, stamina);
    }

    // ==================== System ====================

    /**
     * Get API status/health.
     */
    public ApiStatus getStatus() {
        return new ApiStatus(
            true,
            scheduler.isRunning(),
            profiles.getProfiles().size(),
            System.currentTimeMillis()
        );
    }

    /**
     * Simple status DTO
     */
    public static class ApiStatus {
        public final boolean apiUp;
        public final boolean botRunning;
        public final int profileCount;
        public final long timestamp;

        public ApiStatus(boolean apiUp, boolean botRunning, int profileCount, long timestamp) {
            this.apiUp = apiUp;
            this.botRunning = botRunning;
            this.profileCount = profileCount;
            this.timestamp = timestamp;
        }
    }
}
```

### 2. Update MessageDispatcher

**File:** `wos-api/src/main/java/cl/camodev/wosbot/api/server/MessageDispatcher.java`

```java
package cl.camodev.wosbot.api.server;

import cl.camodev.wosbot.api.protocol.Command;
import cl.camodev.wosbot.api.protocol.Response;
import cl.camodev.wosbot.api.service.BotApiService;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

/**
 * Dispatches incoming commands to BotApiService and returns responses.
 */
public class MessageDispatcher {
    private static final Logger logger = LoggerFactory.getLogger(MessageDispatcher.class);

    private final ObjectMapper objectMapper = new ObjectMapper();
    private final BotApiService apiService = new BotApiService();

    public String handleMessage(String message) {
        try {
            Command command = objectMapper.readValue(message, Command.class);
            logger.info("Handling command: {} (id: {})", command.getCommand(), command.getId());

            Response response = dispatchCommand(command);
            return objectMapper.writeValueAsString(response);

        } catch (Exception e) {
            logger.error("Failed to handle message: {}", message, e);
            return createErrorResponse(null, "Failed to process message: " + e.getMessage());
        }
    }

    private Response dispatchCommand(Command command) {
        try {
            Object result = executeCommand(command);
            return Response.success(command.getId(), result);

        } catch (Exception e) {
            logger.error("Command execution failed: {}", command.getCommand(), e);
            return Response.error(command.getId(), e.getMessage());
        }
    }

    private Object executeCommand(Command command) {
        String cmd = command.getCommand();
        Object params = command.getParams();

        switch (cmd) {
            // Bot control
            case "startBot":
                apiService.startBot();
                return null;

            case "stopBot":
                apiService.stopBot();
                return null;

            case "pauseAllQueues":
                apiService.pauseAllQueues();
                return null;

            case "resumeAllQueues":
                apiService.resumeAllQueues();
                return null;

            case "pauseQueue":
                Long pauseProfileId = extractLongParam(params, "profileId");
                apiService.pauseQueue(pauseProfileId);
                return null;

            case "resumeQueue":
                Long resumeProfileId = extractLongParam(params, "profileId");
                apiService.resumeQueue(resumeProfileId);
                return null;

            // Configuration
            case "getGlobalConfig":
                return apiService.getGlobalConfig();

            case "saveGlobalConfig":
                String key = extractStringParam(params, "key");
                String value = extractStringParam(params, "value");
                apiService.saveGlobalConfig(key, value);
                return null;

            // Profiles
            case "getProfiles":
                return apiService.getProfiles();

            case "getProfileById":
                Long profileId = extractLongParam(params, "profileId");
                return apiService.getProfileById(profileId);

            // Stamina
            case "getCurrentStamina":
                Long staminaProfileId = extractLongParam(params, "profileId");
                return apiService.getCurrentStamina(staminaProfileId);

            // Status
            case "getStatus":
                return apiService.getStatus();

            default:
                throw new IllegalArgumentException("Unknown command: " + cmd);
        }
    }

    // Helper methods to extract parameters
    private Long extractLongParam(Object params, String key) {
        if (params instanceof java.util.Map) {
            Object value = ((java.util.Map<?, ?>) params).get(key);
            if (value instanceof Number) {
                return ((Number) value).longValue();
            }
        }
        throw new IllegalArgumentException("Missing or invalid parameter: " + key);
    }

    private String extractStringParam(Object params, String key) {
        if (params instanceof java.util.Map) {
            Object value = ((java.util.Map<?, ?>) params).get(key);
            if (value != null) {
                return value.toString();
            }
        }
        throw new IllegalArgumentException("Missing parameter: " + key);
    }

    private String createErrorResponse(String id, String error) {
        try {
            return objectMapper.writeValueAsString(Response.error(id, error));
        } catch (Exception e) {
            return "{\"success\":false,\"error\":\"Failed to create error response\"}";
        }
    }
}
```
